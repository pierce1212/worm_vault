```dataviewjs
const { DateTime } = dv.luxon;
const today = DateTime.now().startOf("day");
const sourceFile = dv.current().file.path;
const sourceFolder = sourceFile.includes("/") ? sourceFile.slice(0, sourceFile.lastIndexOf("/")) : "";
const localPath = name => sourceFolder ? `${sourceFolder}/${name}` : name;
const taskSourcePaths = [localPath("工作任务.md")];
const api = typeof require === "function" ? require("obsidian") : {};
const coreModule = { exports: {} };
new Function("module", await app.vault.adapter.read(localPath("tasks-core.js")))(coreModule);
const taskCore = coreModule.exports.create({ app, paths: taskSourcePaths, folder: sourceFolder, DateTime, setIcon: api.setIcon });
await taskCore.initialize();
const style = document.createElement("style");
style.textContent = await taskCore.syncStyles();
dv.container.appendChild(style);
const recordModule = { exports: {} };
new Function("module", await app.vault.adapter.read(localPath("task-records.js")))(recordModule);
const taskRecords = recordModule.exports.create({ app, core: taskCore, DateTime, parseYaml: api.parseYaml, stringifyYaml: api.stringifyYaml, titleOf: coreModule.exports.title });
const errorNode = document.createElement("p");
errorNode.className = "tcc-record-sync-error";
errorNode.setAttribute("role", "alert");
dv.container.appendChild(errorNode);
const report = errors => { errorNode.hidden = !errors.length; errorNode.textContent = errors.join("；"); };
report(await taskRecords.syncAll());
taskRecords.watch(taskSourcePaths, sourceFolder, report);
const todayModule = { exports: {} };
new Function("module", await app.vault.adapter.read(localPath("tasks-today.js")))(todayModule);
const tasks = dv.pages().where(page => taskSourcePaths.includes(page.file.path)).file.tasks.where(task => task.task).array();
const checkinModule = { exports: {} };
new Function("module", await app.vault.adapter.read(localPath("task-checkins.js")))(checkinModule);
const checkins = checkinModule.exports.create({ app, core: taskCore, DateTime, parseYaml: api.parseYaml, stringifyYaml: api.stringifyYaml, titleOf: coreModule.exports.title });
const checkinErrors = await checkins.load(tasks);
if (checkinErrors.length) {
  const warning = document.createElement("p");
  warning.className = "tcc-record-sync-error";
  warning.textContent = checkinErrors.join("；");
  dv.container.appendChild(warning);
}
const groups = todayModule.exports.build(tasks, { core: taskCore, DateTime, today });
const esc = value => String(value ?? "").replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
const array = value => Array.isArray(value) ? value : value?.array?.() ?? Array.from(value ?? []);

function taskTitle(task) {
  return `<button type="button" class="tcc-today-title" data-today-detail="${esc(taskCore.identity(task))}">${esc(coreModule.exports.title(task.text))}</button>`;
}

function childRows(task) {
  const render = items => array(items).map(child => {
    const nested = render(child.children);
    if (!child.task) return nested;
    return `<li><div class="tcc-today-child">
      <input type="checkbox" data-today-check="${esc(taskCore.identity(child))}" ${child.completed ? "checked" : ""} aria-label="完成子任务：${esc(coreModule.exports.title(child.text))}">
      ${taskTitle(child)}${taskCore.statusBadge(child, true)}
    </div>${nested ? `<ul>${nested}</ul>` : ""}</li>`;
  }).join("");
  const kids = taskCore.children(task);
  return kids.length ? `<details class="tcc-today-children"><summary>子任务 ${kids.filter(item => item.completed).length}/${kids.length}</summary><ul>${render(task.children)}</ul></details>` : "";
}

function row(item, section) {
  const task = item.task, progress = taskCore.completion(task), data = taskCore.info(task);
  const due = task.text.match(/📅\s*(\d{4}-\d{2}-\d{2})/u)?.[1];
  return `<article class="tcc-today-row" data-today-task="${esc(taskCore.identity(task))}">
    <input type="checkbox" data-today-check="${esc(taskCore.identity(task))}" ${task.completed ? "checked" : ""} aria-label="完成任务：${esc(coreModule.exports.title(task.text))}">
    <div class="tcc-today-main">${taskTitle(task)}
      <div class="tcc-today-meta"><span>${esc(item.labels.join(" · "))}</span>${due ? `<span>截止 ${due}</span>` : ""}<span>${progress.percent}%${taskCore.children(task).length ? ` · 子任务 ${progress.detail}` : ""}</span></div>
      ${data.next && !task.completed ? `<p class="tcc-today-next">${esc(data.next)}</p>` : ""}
      ${childRows(task)}
    </div>
    <div class="tcc-today-actions"><span class="tcc-priority ${item.priority.cls}">${item.priority.label}${item.childPriority ? " · 子任务" : ""}</span>
      ${taskCore.statusBadge(task, true)}
      ${section === "overdue" ? `<button type="button" data-plan-today="${esc(taskCore.identity(task))}" title="安排今天" aria-label="安排今天"><span data-tcc-icon="calendar-plus">安排今天</span></button>` : ""}
      <button type="button" data-task-action="edit" data-task-key="${esc(taskCore.identity(task))}" title="编辑任务" aria-label="编辑任务"><span data-tcc-icon="pencil">编辑</span></button>
    </div>
  </article>`;
}

const root = document.createElement("section");
root.className = "tcc-today-shell";
root.innerHTML = `<header class="tcc-today-header"><div><h2>今日任务</h2><p>${today.setLocale("zh-cn").toFormat("yyyy-LL-dd cccc")}</p></div>
  <nav><button type="button" data-task-action="create" data-create-scheduled="${today.toFormat("yyyy-LL-dd")}"><span data-tcc-icon="plus"></span>新增今日任务</button><button type="button" data-habit-achievements>打卡成就</button><button type="button" data-today-dashboard>任务中心</button></nav></header>
  ${checkins.todayHtml(groups.habits)}
  <div class="tcc-today-summary"><span>今日 ${groups.planned.length} 项</span><span>已完成 ${groups.done.length} 项</span><span>优先级从高到低</span></div>
  <div class="tcc-today-list" data-today-list>${groups.planned.map(item => row(item, "planned")).join("") || '<p class="tcc-empty">今天暂无安排</p>'}</div>
  <details class="tcc-today-secondary"><summary>逾期待安排 <span>${groups.overdue.length}</span></summary><div>${groups.overdue.map(item => row(item, "overdue")).join("") || '<p class="tcc-empty">暂无逾期待办</p>'}</div></details>
  <details class="tcc-today-secondary"><summary>今日已完成 <span>${groups.done.length}</span></summary><div>${groups.done.map(item => row(item, "done")).join("") || '<p class="tcc-empty">今天还没有完成记录</p>'}</div></details>`;
dv.container.appendChild(root);
taskCore.bind(root, tasks);
checkins.bind(root, tasks, sourceFile);
root.addEventListener("click", async event => {
  if (event.target.closest("[data-today-dashboard]")) app.workspace.openLinkText(localPath("obsidian_tasks_command_center.md"), sourceFile, false);
  const detail = event.target.closest("[data-today-detail]");
  if (detail) {
    window.localStorage.setItem("tcc:selectedTaskId", detail.dataset.todayDetail);
    app.workspace.openLinkText(localPath("任务详情.md"), sourceFile, false);
  }
  const plan = event.target.closest("[data-plan-today]");
  if (!plan || plan.disabled) return;
  const task = tasks.find(item => taskCore.identity(item) === plan.dataset.planToday);
  if (!task) return;
  plan.disabled = true;
  try {
    const ok = await taskCore.attempt(() => taskCore.save(task, { scheduled: today.toFormat("yyyy-LL-dd") }), root);
    if (ok) await taskCore.refresh();
  } finally { plan.disabled = false; }
});
root.addEventListener("change", async event => {
  const input = event.target.closest("[data-today-check]");
  if (!input) return;
  const task = tasks.find(item => taskCore.identity(item) === input.dataset.todayCheck);
  if (!task) return;
  input.disabled = true;
  try {
    const ok = await taskCore.attempt(() => taskCore.save(task, { status: input.checked ? "x" : " " }), root);
    if (ok) await taskCore.refresh();
    else input.checked = task.completed;
  } finally { input.disabled = false; }
});
```
