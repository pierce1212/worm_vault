```dataviewjs
const { DateTime } = dv.luxon;
const today = DateTime.now().startOf("day");
const sourceFile = dv.current().file.path;
const sourceFolder = sourceFile.slice(0, sourceFile.lastIndexOf("/"));
const localPath = name => sourceFolder ? `${sourceFolder}/${name}` : name;
const taskSourcePaths = [localPath("工作任务.md")];
const api = typeof require === "function" ? require("obsidian") : {};
const coreModule = { exports: {} };
new Function("module", await app.vault.adapter.read(localPath("tasks-core.js")))(coreModule);
const core = coreModule.exports.create({ app, paths: taskSourcePaths, folder: sourceFolder, DateTime, setIcon: api.setIcon });
await core.initialize();
const style = document.createElement("style");
style.textContent = await core.syncStyles();
dv.container.appendChild(style);
const module = { exports: {} };
new Function("module", await app.vault.adapter.read(localPath("task-checkins.js")))(module);
const checkins = module.exports.create({ app, core, DateTime, parseYaml: api.parseYaml, stringifyYaml: api.stringifyYaml, titleOf: coreModule.exports.title });
const tasks = dv.pages().where(page => taskSourcePaths.includes(page.file.path)).file.tasks.where(task => task.task).array();
await checkins.load(tasks);
const data = checkins.overview(tasks);
const esc = value => String(value ?? "").replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
const root = document.createElement("section");
root.className = "tcc-today-shell tcc-achievement-shell";
root.innerHTML = `<header class="tcc-today-header"><div><h2>打卡成就</h2><p>${today.toFormat("yyyy-LL-dd")} · 每次有效完成 +10 成长值</p></div><nav><button type="button" data-return-today>今日任务</button><button type="button" data-return-center>任务中心</button></nav></header>
  ${data.errors.length ? `<details class="tcc-record-sync-error"><summary>${data.errors.length} 份记录读取不完整</summary><p>${esc(data.errors.join("；"))}</p></details>` : ""}
  <div class="tcc-achievement-totals"><span><strong>${data.growth.count}</strong>累计完成（次）</span><span><strong>${data.growth.activeDays}</strong>有积累的日子</span><span><strong>${data.growth.minutes}</strong>累计分钟</span><span><strong>${data.growth.weekCount}</strong>本周完成（次）</span></div>
  ${checkins.growthHtml(data.growth)}
  <details class="tcc-habit-breakdown"><summary>各习惯积累 ${data.habits.length}</summary><div class="tcc-habit-breakdown-list">${data.habits.map(item => `<article><div>${item.task ? `<button type="button" class="tcc-today-title" data-achievement-task="${esc(item.key)}">${esc(item.name)}</button>` : `<strong>${esc(item.name)}</strong>`}<small>${item.task && core.habit(item.task).enabled ? `本周 ${item.growth.weekCount}/${item.goal} 天` : "历史记录"}</small></div><span>累计 ${item.growth.count} 次 · ${item.growth.minutes} 分钟</span><b>Lv. ${item.growth.level}</b></article>`).join("") || '<p class="tcc-empty">还没有打卡习惯</p>'}</div></details>
  <section class="tcc-achievement-history"><h3>完成记录</h3><div class="tcc-history-filters"><label>习惯<select data-history-task aria-label="筛选习惯"><option value="">全部习惯</option>${data.habits.map(item => `<option value="${esc(item.key)}">${esc(item.name)}</option>`).join("")}</select></label><label>范围<select data-history-range aria-label="筛选范围"><option value="week">本周</option><option value="month">本月</option><option value="all">全部</option></select></label><label>日期<input type="date" data-history-day max="${today.toFormat("yyyy-LL-dd")}" aria-label="筛选日期"></label><label class="tcc-history-rest"><input type="checkbox" data-history-rest>包含休息日</label><button type="button" data-history-reset title="重置筛选" aria-label="重置筛选"><span data-tcc-icon="rotate-ccw">重置</span></button></div><p class="tcc-muted" data-history-count></p><div data-history-list></div><button type="button" data-history-more hidden>显示更多</button></section>`;
dv.container.appendChild(root);
checkins.bind(root, tasks, sourceFile);
core.icons(root);
let limit = 50;
function renderHistory() {
  const task = root.querySelector("[data-history-task]").value;
  const range = root.querySelector("[data-history-range]").value;
  const date = root.querySelector("[data-history-day]").value;
  const rest = root.querySelector("[data-history-rest]").checked;
  const from = (range === "week" ? today.startOf("week") : range === "month" ? today.startOf("month") : null)?.toFormat("yyyy-LL-dd");
  const entries = data.entries.filter(entry => (!task || entry.taskId === task) && (!date || entry.day === date) && (date || !from || entry.day >= from) && (rest || entry.kind === "done"));
  root.querySelector("[data-history-count]").textContent = `${entries.length} 条记录 · ${entries.filter(entry => entry.kind === "done").reduce((sum, entry) => sum + entry.minutes, 0)} 分钟`;
  let previousDay = "";
  root.querySelector("[data-history-list]").innerHTML = entries.slice(0, limit).map(entry => {
    const heading = entry.day !== previousDay ? `<h4>${entry.day}</h4>` : "";
    previousDay = entry.day;
    return `${heading}<article class="tcc-achievement-entry"><div><strong>${esc(entry.name)}</strong><span>${entry.kind === "done" ? `${entry.minutes} 分钟 · 已获得 10 成长值` : "休息日"}</span><p>${esc(entry.reflection || entry.content || entry.next || "")}</p></div><div class="tcc-achievement-entry-actions">${entry.task && core.habit(entry.task).enabled ? `<button type="button" data-habit-checkin="${esc(entry.taskId)}" data-checkin-day="${entry.day}">修改</button>` : ""}<button type="button" data-habit-path="${esc(entry.path)}">月度记录</button>${(entry.notePaths ?? []).map(path => `<button type="button" data-habit-note-path="${esc(path)}">${esc(path.split("/").at(-1).replace(/\.md$/i, ""))}</button>`).join("")}</div></article>`;
  }).join("") || '<p class="tcc-empty">这个范围内还没有完成记录</p>';
  root.querySelector("[data-history-more]").hidden = entries.length <= limit;
}
root.querySelectorAll("[data-history-task], [data-history-range], [data-history-day], [data-history-rest]").forEach(input => input.addEventListener("change", () => { limit = 50; renderHistory(); }));
root.addEventListener("click", event => {
  const target = event.target;
  if (target.closest("[data-return-today]")) app.workspace.openLinkText(localPath("今日任务.md"), sourceFile, false);
  if (target.closest("[data-return-center]")) app.workspace.openLinkText(localPath("obsidian_tasks_command_center.md"), sourceFile, false);
  const task = target.closest("[data-achievement-task]");
  if (task) { window.localStorage.setItem("tcc:selectedTaskId", task.dataset.achievementTask); app.workspace.openLinkText(localPath("任务详情.md"), sourceFile, false); }
  const day = target.closest("[data-reward-day]");
  if (day) { root.querySelector("[data-history-day]").value = day.dataset.rewardDay; root.querySelector("[data-history-rest]").checked = true; limit = 50; renderHistory(); }
  if (target.closest("[data-history-reset]")) { root.querySelector("[data-history-task]").value = ""; root.querySelector("[data-history-day]").value = ""; root.querySelector("[data-history-range]").value = "week"; root.querySelector("[data-history-rest]").checked = false; limit = 50; renderHistory(); }
  if (target.closest("[data-history-more]")) { limit += 50; renderHistory(); }
});
renderHistory();
```
