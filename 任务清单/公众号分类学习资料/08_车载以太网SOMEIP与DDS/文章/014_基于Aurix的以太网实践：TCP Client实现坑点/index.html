1、TCP Client实现坑点

Ethernet通信，需要收/发两个节点，我们习惯称为Client和Server。相对Server端实现，Client端实现需要注意更多的细节。先说一下本文的环境：
  • 软件：Lwip，无操作系统
  • 硬件：TC375、TC397
本文TCP通信的拓扑环境如下所示：

（一）Server、Client启动时序

实现过程中，先启动Server端，再启动Client端。即：让Server端先进入Listen（监听）模式，之后再让Client请求链接。

Client端启动时，需要确保网卡已经Link Up，否则，可能导致Client连接服务器失败。本文的处理如下（Cpu0_Main.c）：

......while (1){    if((g_Lwip.netif.flags & NETIF_FLAG_LINK_UP) && (LinkUpFlag == FALSE))    {        LinkUpFlag = TRUE;        tcp_client_init();    }    Ifx_Lwip_pollTimerFlags();       Ifx_Lwip_pollReceiveFlags();}......
（二）服务端接口实现
服务端实现的接口如下所示：
void tcp_client_init(void){  #define TCP_REMOTE_PORT  8088  #define TCP_LOCAL_PORT   8088
  struct tcp_pcb *tpcb;   /* 服务器IP */   ip_addr_t serverIp;   IP4_ADDR(&serverIp, 192, 168, 0, 11);
   /* 创建tcp控制块 */   tpcb = tcp_new();      if (tpcb != NULL)   {       err_t err;              err = tcp_bind(tpcb, IP_ADDR_ANY, TCP_LOCAL_PORT);
       if (err == ERR_OK)       {           /* 连接服务器 */           tcp_connect(tpcb, &serverIp, TCP_REMOTE_PORT, tcp_client_connected);       }       else       {           memp_free(MEMP_TCP_PCB, tpcb);       }   }}
static err_t tcp_client_connected(void *arg, struct tcp_pcb *tpcb, err_t err){    LWIP_UNUSED_ARG(arg);    LWIP_UNUSED_ARG(err);    LWIP_DEBUGF(NETIF_DEBUG | LWIP_DBG_TRACE | LWIP_DBG_STATE, ("tcp_client_connected\r\n"));
    /* 注册接收回调函数 */    tcp_recv(tpcb, tcp_client_recv);    tcp_poll(tpcb, tcp_poll_send, 1);        tcp_output(tpcb);}
static err_t tcp_client_recv(void *arg, struct tcp_pcb *tpcb,                            struct pbuf *p, err_t err){    LWIP_DEBUGF(NETIF_DEBUG | LWIP_DBG_TRACE | LWIP_DBG_STATE, ("tcp_client_recv\r\n"));
    if (p != NULL)    {        struct pbuf *ptmp = p;                while(ptmp != NULL)        {            ptmp = p->next;        }                tcp_recved(tpcb, p->tot_len);                /* 释放缓冲区数据 */        pbuf_free(p);    }    else if (err == ERR_OK)    {        LWIP_DEBUGF(NETIF_DEBUG | LWIP_DBG_TRACE | LWIP_DBG_STATE, ("tcp client close\r\n"));        tcp_recved(tpcb, p->tot_len);                return tcp_close(tpcb);    }
    return ERR_OK;}
static err_t tcp_poll_send(void *arg, struct tcp_pcb *tpcb){    #define TCP_SEND_LEN       200
    tcp_nagle_disable(tpcb);    tcp_write(tpcb, "tcp_client", TCP_SEND_LEN, TCP_WRITE_FLAG_COPY);    tcp_output(tpcb);    return ERR_OK; }

解释：

使用tcp_poll(struct tcp_pcb *pcb, tcp_poll_fn poll, u8_t interval)接口注册一个周期callback function，在此回调中周期发送TCP数据。interval表示间隔，本文设置1，表示200ms。对于TCP的定时器间隔，可以修改tcp_priv.h文件中的宏：TCP_TMR_INTERVAL，eg：50。修改如下所示：

（三）ARP处理
